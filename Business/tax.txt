        new SessionLineItemOptions
        {
            PriceData = new SessionLineItemPriceDataOptions
            {
                UnitAmount = 10000, // $100.00
                TaxCode = "txcd_50010003", // Concert category

            {
                TaxDetails = new SessionLineItemProductDataTaxDetailsOptions
                {
                    PerformanceLocation = new SessionLineItemProductDataTaxDetailsPerformanceLocationOptions
                    {
                        Address = new AddressOptions
                        {
                            Line1 = "123 Venue Street",
                            City = "Denver",
                            State = "CO",
                            PostalCode = "80202",
                            Country = "US",
                        },
                    },
                },
            },
        },
    },
    AutomaticTax = new SessionAutomaticTaxOptions { Enabled = true },
