 Your .net core API can fetch the real-time status directly from Stripe using the organizer's account ID and return a simple true/false
 checklist to your React view.Add an endpoint like /api/organizer/stripe-status/{accountId} 
 to your backend:csharp[HttpGet("api/organizer/stripe-status/{accountId}")]
public async Task<IActionResult> GetStripeStatus(string accountId)
{
    // Initialize using your platform secret key (sk_test_... or sk_live_...)
    var stripeClient = new StripeClient(_configuration["Stripe:SecretKey"]);
    var accountService = new AccountService(stripeClient);

    try
    {
        // Fetch the connected account details directly from Stripe's API
        Account account = await accountService.GetAsync(accountId);

        var onboardingStatus = new
        {
            IsConnected = true,
            // ChargesEnabled must be true for Stripe Checkout to work
            CanAcceptPayments = account.ChargesEnabled, 
            // PayoutsEnabled must be true for them to receive their cash
            CanReceivePayouts = account.PayoutsEnabled,
            // If false, they still have fields to fill out on Stripe's dashboard
            IsDetailsSubmitted = account.DetailsSubmitted 
        };

        return Ok(onboardingStatus);
    }
    catch (StripeException ex)
    {
        return BadRequest(new { message = "Failed to fetch Stripe account info.", error = ex.Message });
    }
}
